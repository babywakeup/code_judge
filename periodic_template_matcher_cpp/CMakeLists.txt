cmake_minimum_required(VERSION 3.20)
project(periodic_template_matcher_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PTM_BUILD_PYTHON "Build Python array binding" ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc photo video)
find_package(Threads REQUIRED)
file(GLOB PTM_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(ptm_core STATIC ${PTM_SOURCES})
target_include_directories(ptm_core PUBLIC include ${OpenCV_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}/.deps/include")
target_link_libraries(ptm_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
if(MSVC)
  target_compile_options(ptm_core PRIVATE /W4)
else()
  target_compile_options(ptm_core PRIVATE -Wall -Wextra)
endif()
add_executable(ptm_debug debug/main.cpp)
target_link_libraries(ptm_debug PRIVATE ptm_core)
if(PTM_BUILD_PYTHON)
  set(PYBIND11_FINDPYTHON ON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_native NO_EXTRAS bindings/module.cpp)
  target_link_libraries(_native PRIVATE ptm_core)
  # Keep configurations independent. build.py publishes the chosen artifact.
  set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/python" RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/python")
  endforeach()
endif()
enable_testing()
add_test(NAME native_smoke COMMAND ptm_debug)
add_executable(ptm_native_contract tests/native_contract.cpp)
target_link_libraries(ptm_native_contract PRIVATE ptm_core)
add_test(NAME native_input_contract COMMAND ptm_native_contract)
